## *优化后的 ReAct 循环*

for step in range(max_steps):   把输入输出  定义好确定  ai/手写
    # 1. 推理
    thought = llm.think(context, last_obs)
    action = llm.decide_action(thought)

    # 2. 行动
    result = execute_action(action)

    # 3. 观察
    observation = observe_scene()

    # 4. 反思 (新增)
    if not observation["ok"]:
        reflection = llm.reflect(
            thought=thought,
            action=action,
            observation=observation,
            context=context
        )
        # reflection 可能是:
        # "我放置 Box 在 Conveyor 上，但它超出了边界。
        #  原因是我没有考虑 Conveyor 的实际尺寸。
        #  下次应该先 query_scene 获取精确尺寸。"

    context["reflection"] = reflection
        context["lessons_learned"].append(reflection)

    # 5. 策略调整 (新增)
    if needs_strategy_change(observation, reflection):
        strategy = llm.adjust_strategy(context)
        context["current_strategy"] = strategy

# 优化方向5 引入仿真

## 优化后: 每次放置后运行物理仿真

def observe_with_simulation(layout):
    # 1. 静态验证
    static_validation = validate_layout(layout)

    # 2. 物理仿真
    simulation = run_physics_simulation(layout, duration=5.0)

    # 3. 提取仿真反馈
    physics_feedback = {
        "stable_assets": simulation.stable_objects,
        "fallen_assets": simulation.fallen_objects,不稳定资产
        "contacts": simulation.contact_points,  接触点
        "warnings": simulation.warnings,
    }

    return {
        "static_validation": static_validation,
        "physics_feedback": physics_feedback,
    }
